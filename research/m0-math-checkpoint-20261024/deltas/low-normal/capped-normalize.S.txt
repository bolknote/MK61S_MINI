.Lda_low_near_normalize:
 lsrs r5,r1,#20
 bne .Lda_low_near_pack
 // R7 is the number of exact left shifts available before exponent1.
 lsls r7,r4,#21
 lsrs r7,#21
 cmp r7,#1
 beq .Lda_low_near_pack
 subs r7,#1
 lsrs r5,r1,#16
 bne .Lda_low_norm_one
.Lda_low_norm_sixteen:
 cmp r1,#16
 bhs .Lda_low_norm_eight
 cmp r7,#16
 blo .Lda_low_norm_short
 lsls r1,#16
 lsrs r5,r0,#16
 orrs r1,r5
 lsls r0,#16
 subs r4,#16
 subs r7,#16
 b .Lda_low_norm_sixteen
.Lda_low_norm_eight:
 lsrs r5,r1,#12
 bne .Lda_low_norm_four
 cmp r7,#8
 blo .Lda_low_norm_short
 lsls r1,#8
 lsrs r5,r0,#24
 orrs r1,r5
 lsls r0,#8
 subs r4,#8
 subs r7,#8
.Lda_low_norm_four:
 lsrs r5,r1,#16
 bne .Lda_low_norm_one
 cmp r7,#4
 blo .Lda_low_norm_short
 lsls r1,#4
 lsrs r5,r0,#28
 orrs r1,r5
 lsls r0,#4
 subs r4,#4
 subs r7,#4
.Lda_low_norm_one:
 cmp r7,#0
 beq .Lda_low_near_pack
 adds r0,r0
 adcs r1,r1
 subs r4,#1
 subs r7,#1
 lsrs r5,r1,#20
 beq .Lda_low_norm_one
 b .Lda_low_near_pack
.Lda_low_norm_short:
 // Needed normalization exceeds the remaining budget. Shift exactly to
 // exponent1 and pack the raw subnormal coefficient without discarding bits.
 cmp r7,#0
 beq .Lda_low_near_pack
 movs r5,#32
 subs r5,r5,r7
 movs r6,r0
 lsrs r6,r5
 lsls r1,r7
 orrs r1,r6
 lsls r0,r7
 subs r4,r4,r7
 b .Lda_low_near_pack

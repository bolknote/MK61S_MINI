name=MK61s F401/F411
version=1.1.1

# The rest of platform.txt is inherited from the referenced
# STMicroelectronics:arduino core. These hooks add MK61s F401/F411 profile
# validation and System APP packaging.
recipe.hooks.prebuild.20.pattern="{runtime.platform.path}/tools/mk61-app-postbuild.sh" check-profile --platform "{build.mk61_platform_id}" --display "{build.mk61_display_id}" --usb-screen "{build.mk61_usb_screen}" --chip8 "{build.mk61_chip8}" --sketch "{build.source.path}" --build-path "{build.path}" --variant-ld "{build.variant.path}/{build.ldscript}"
recipe.hooks.prebuild.20.pattern.windows=powershell.exe -NoLogo -NoProfile -ExecutionPolicy Bypass -File "{runtime.platform.path}/tools/mk61-app-postbuild.ps1" check-profile -Platform "{build.mk61_platform_id}" -Display "{build.mk61_display_id}" -UsbScreen "{build.mk61_usb_screen}" -Chip8 "{build.mk61_chip8}" -Sketch "{build.source.path}" -BuildPath "{build.path}" -VariantLd "{build.variant.path}/{build.ldscript}"

recipe.hooks.objcopy.postobjcopy.20.pattern="{runtime.platform.path}/tools/mk61-app-postbuild.sh" build --compiler "{compiler.path}{compiler.cpp.cmd}" --build-path "{build.path}" --sketch "{build.source.path}" --project "{build.project_name}" --bundle "mk61s-M-{build.mk61_platform_id}-{build.mk61_display_id}-{build.mk61_mcu_id}" --mcu "{build.mk61_mcu_id}" --max-size "{upload.maximum_size}" --focal "{build.mk61_focal_app}" --basic "{build.mk61_basic_app}" --wbmp "{build.mk61_wbmp_app}" --markdown "{build.mk61_markdown_app}" --chip8 "{build.mk61_chip8_app}" --setup "{build.mk61_setup_app}" --usbdisk "{build.mk61_usbdisk_app}" --explorer "{build.mk61_explorer_app}" --local-float-math "{build.mk61_app_math}" --compile-flags "{build.st_extra_flags}"
recipe.hooks.objcopy.postobjcopy.20.pattern.windows=powershell.exe -NoLogo -NoProfile -ExecutionPolicy Bypass -File "{runtime.platform.path}/tools/mk61-app-postbuild.ps1" build -Compiler "{compiler.path}{compiler.cpp.cmd}" -BuildPath "{build.path}" -Sketch "{build.source.path}" -Project "{build.project_name}" -Bundle "mk61s-M-{build.mk61_platform_id}-{build.mk61_display_id}-{build.mk61_mcu_id}" -Mcu "{build.mk61_mcu_id}" -MaxSize "{upload.maximum_size}" -Focal "{build.mk61_focal_app}" -Basic "{build.mk61_basic_app}" -Wbmp "{build.mk61_wbmp_app}" -Markdown "{build.mk61_markdown_app}" -Chip8 "{build.mk61_chip8_app}" -Setup "{build.mk61_setup_app}" -UsbDisk "{build.mk61_usbdisk_app}" -Explorer "{build.mk61_explorer_app}" -LocalFloatMath "{build.mk61_app_math}" -CompileFlags "{build.st_extra_flags}"

# Arduino has build hooks but no post-Upload hook. Use a board-private upload
# tool name instead of overriding STM32duino's shared stm32CubeProg recipe:
# platform inheritance order must never be able to bypass APP installation.
# Non-Windows hosts retain the ordinary resident-only DFU command. Windows
# flashes resident and then installs this build's System APP through the MKC
# CDC protocol (USB disk is not required).
tools.mk61Upload.upload.params.verbose=
tools.mk61Upload.upload.params.quiet=
tools.mk61Upload.upload.pattern="{busybox}" sh "{runtime.tools.STM32Tools.path}/stm32CubeProg.sh" -i {upload.protocol} -f "{build.path}/{build.project_name}.bin" -o {build.flash_offset} {upload.options}
tools.mk61Upload.upload.pattern.windows=powershell.exe -NoLogo -NoProfile -ExecutionPolicy Bypass -File "{runtime.platform.path}/tools/mk61-app-upload.ps1" -Busybox "{busybox}" -Stm32Script "{runtime.tools.STM32Tools.path}/stm32CubeProg.sh" -BuildPath "{build.path}" -Project "{build.project_name}" -Bundle "mk61s-M-{build.mk61_platform_id}-{build.mk61_display_id}-{build.mk61_mcu_id}" -Profile "{build.mk61_platform_id}-{build.mk61_display_id}" -Mcu "{build.mk61_mcu_id}" -RequireUsbDisk "{build.mk61_usbdisk_app}" -Protocol "{upload.protocol}" -FlashOffset "{build.flash_offset}" -Vid "{upload.vid}" -UsbPid "{upload.pid}" -Address "{upload.address}" -Start "{upload.start}" -Port "{serial.port}"

compiler.c.elf.extra_flags="-Wl,--default-script={build.path}/mk61-portable.ld"

name=MK61s F401 + APP
version=1.0.0

# The rest of platform.txt is inherited from the referenced
# STMicroelectronics:arduino core.  These hooks add only MK61s profile
# validation and System APP packaging.
recipe.hooks.prebuild.20.pattern="{runtime.platform.path}/tools/mk61-app-postbuild.sh" check-profile --platform "{build.mk61_platform_id}" --display "{build.mk61_display_id}" --sketch "{build.source.path}" --build-path "{build.path}" --variant-ld "{build.variant.path}/{build.ldscript}"
recipe.hooks.prebuild.20.pattern.windows=powershell.exe -NoLogo -NoProfile -ExecutionPolicy Bypass -File "{runtime.platform.path}/tools/mk61-app-postbuild.ps1" check-profile -Platform "{build.mk61_platform_id}" -Display "{build.mk61_display_id}" -Sketch "{build.source.path}" -BuildPath "{build.path}" -VariantLd "{build.variant.path}/{build.ldscript}"

recipe.hooks.objcopy.postobjcopy.20.pattern="{runtime.platform.path}/tools/mk61-app-postbuild.sh" build --compiler "{compiler.path}{compiler.cpp.cmd}" --build-path "{build.path}" --sketch "{build.source.path}" --project "{build.project_name}" --bundle "mk61s-M-{build.mk61_platform_id}-{build.mk61_display_id}-f401" --focal "{build.mk61_focal}" --basic "{build.mk61_basic}" --wbmp "{build.mk61_wbmp}" --markdown "{build.mk61_markdown}" --chip8 "{build.mk61_chip8}" --local-float-math "{build.mk61_app_math}" --compile-flags "{build.st_extra_flags}"
recipe.hooks.objcopy.postobjcopy.20.pattern.windows=powershell.exe -NoLogo -NoProfile -ExecutionPolicy Bypass -File "{runtime.platform.path}/tools/mk61-app-postbuild.ps1" build -Compiler "{compiler.path}{compiler.cpp.cmd}" -BuildPath "{build.path}" -Sketch "{build.source.path}" -Project "{build.project_name}" -Bundle "mk61s-M-{build.mk61_platform_id}-{build.mk61_display_id}-f401" -Focal "{build.mk61_focal}" -Basic "{build.mk61_basic}" -Wbmp "{build.mk61_wbmp}" -Markdown "{build.mk61_markdown}" -Chip8 "{build.mk61_chip8}" -LocalFloatMath "{build.mk61_app_math}" -CompileFlags "{build.st_extra_flags}"

# Arduino has build hooks but no post-Upload hook. Replace only this board's
# Windows DFU recipe: flash resident, then install this build's System APP
# through the existing MKC CDC file protocol (USB disk is not required).
tools.stm32CubeProg.upload.pattern.windows=powershell.exe -NoLogo -NoProfile -ExecutionPolicy Bypass -File "{runtime.platform.path}/tools/mk61-app-upload.ps1" -Busybox "{busybox}" -Stm32Script "{runtime.tools.STM32Tools.path}/stm32CubeProg.sh" -BuildPath "{build.path}" -Project "{build.project_name}" -Bundle "mk61s-M-{build.mk61_platform_id}-{build.mk61_display_id}-f401" -Profile "{build.mk61_platform_id}-{build.mk61_display_id}" -Sketch "{build.source.path}" -Protocol "{upload.protocol}" -FlashOffset "{build.flash_offset}" -Vid "{upload.vid}" -UsbPid "{upload.pid}" -Address "{upload.address}" -Start "{upload.start}" -Port "{serial.port}"

compiler.c.elf.extra_flags="-Wl,--default-script={build.path}/mk61-portable.ld"
